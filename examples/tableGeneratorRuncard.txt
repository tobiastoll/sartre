#
# DFT/FFT-based table filling (nuclear targets, A > 1 only).
#
# useArbitraryTGrid = true uses the DFT option, otherwise FFT option, useFFT = true in both cases
#
# useFFT = true:
#   The b-space integral is computed via 2D FFT, delivering amplitudes at
#   ALL Delta values simultaneously.  The outer loop runs over (Q2, W2)
#   rows only (nQ2 x nW2 rows instead of nQ2 x nW2 x nt bins), and the
#   t-axis of the table is set automatically to the FFT Delta grid:
#       t_n = -(n * dDelta)^2,  n = 1 ... nFFT/2
#   with variable-width bins centred exactly at each t_n.
#   The 'tbins', 'tmin', 'tmax' settings are ignored in FFT mode.
#   startBin / endBin are interpreted as (Q2 x W2) row indices (0-indexed,
#   range 0 ... Q2bins*W2bins-1).
#   modesToCalculate is effectively overridden to 2 (numerical coherent +
#   incoherent); lambda tables are not produced.
#
# nFFT sets the FFT grid size N (must be even; default 64).
#   A larger N gives more t-bins AND larger |t| coverage, because
#   dDelta = 2*pi*hbarc / (N*db) is fixed by bmax (independent of N),
#   so |t|_max = (N/2)^2 * dDelta^2 grows quadratically with N.
#   For Pb (bmax ~ 17.5 fm):  N=64 -> |t|_max ~ 1.3 GeV^2
#                              N=128 -> |t|_max ~ 5.1 GeV^2
#                              N=256 -> |t|_max ~ 20 GeV^2
#   With hotspots (hasSubstructure=true) use N >= 256: the hotspot scale
#   (~0.2-0.3 fm) requires db < 0.2 fm, which needs N > 2*bmax/0.2 ~ 175.
#
# userInt defines the region in |t|:
#       |t|<0.04: Region 1, nFFT=128, numberOfConfigurations=200, 3, hasSubstructure = false, second dip
#  0.04<|t|<0.4 : Region 2, nFFT=256, numberOfConfigurations=500, 2.5, hasSubstructure = true, nucleon fluctuations
#   0.4<|t|<2.5 : Region 3, nFFT=256, numberOfConfigurations=600, 2.2, hasSubstructure = true, hotspot fluctuations
#
useArbitraryTGrid = true
useFFT = false
userInt = 1
nFFT   = 128
bmaxFactorFFT = 3.0
hasSubstructure = true

# distributes configurations across threads:
#
#   nThreads = 1   serial — no overhead, works everywhere (default)
#   nThreads = N   use exactly N threads
#   nThreads = 0   use all available hardware threads
#                  (equivalent to setting OMP_NUM_THREADS in the shell)
#
# Requirements:
#   • Sartre must be compiled with OpenMP:  cmake -DENABLE_OMP=ON ..
#   • If OpenMP was not compiled in, this setting is silently ignored.
#
# Recommended starting point: set nThreads to the number of physical cores,
# e.g. 8 for an 8-core machine.  Hyperthreading rarely helps here because
# the FFT and dipole model evaluations are compute-bound.
#
# Note: setting nThreads > nConfigurations gives no benefit — there is at
# most one thread per configuration.
#
nThreads = 0
